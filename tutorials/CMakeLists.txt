cmake_minimum_required(VERSION 3.24)
project(queue_tutorial_witness LANGUAGES CXX)
find_package(Threads REQUIRED)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(Q1_ENABLE_TSAN "Instrument the standalone witness with ThreadSanitizer" OFF)
add_executable(q1_passing_an_item examples/q1_passing_an_item.cpp)
target_include_directories(q1_passing_an_item PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../include")
target_compile_features(q1_passing_an_item PRIVATE cxx_std_20)
target_link_libraries(q1_passing_an_item PRIVATE Threads::Threads)
target_compile_options(q1_passing_an_item PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
if(Q1_ENABLE_TSAN)
  target_compile_options(q1_passing_an_item PRIVATE -fsanitize=thread -fno-omit-frame-pointer)
  target_link_options(q1_passing_an_item PRIVATE -fsanitize=thread)
endif()
enable_testing()
add_test(NAME q1_passing_an_item COMMAND q1_passing_an_item)
set_tests_properties(q1_passing_an_item PROPERTIES TIMEOUT 10)
